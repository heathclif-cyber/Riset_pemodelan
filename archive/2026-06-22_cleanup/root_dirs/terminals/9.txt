def apply_hmm(p0,p2,hmm):
    n=len(p0); tl=np.full(n,0.45); ts=np.full(n,0.45)
    for s,(a,b) in HMM.items():
        if s==-1: continue
        m=hmm==s; tl[m]=a; ts[m]=b
    y=np.ones(n,dtype=np.int32)
    lm=p2>=tl; y[lm]=2
    sm=(p0>=ts)&~lm; y[sm]=0
    return y

def cs_apply(group, k, min_p2=0.0, min_p0=0.0):

    g['cs_long'] = False; g['cs_short'] = False
